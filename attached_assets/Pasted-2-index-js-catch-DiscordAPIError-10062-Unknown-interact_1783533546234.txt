2. แก้ index.js catch กลาง

เป้าหมาย: ลดปัญหา DiscordAPIError[10062]: Unknown interaction และ warning ephemeral deprecated

ตอนนี้ index.js catch กลางยังใช้ ephemeral: true อยู่ ให้แก้เป็น flags: 64 และ ignore error ที่ไม่ควร spam log

เพิ่ม helper ใน bot/src/index.js:

function isIgnorableInteractionError(error) {
  return error?.code === 10062 || error?.code === 40060;
}

function logDiscordError(label, error) {
  console.error(label, {
    code: error?.code,
    status: error?.status,
    message: error?.message,
    method: error?.method,
  });
}

แล้วแก้ catch ใน interactionCreate จากเดิมเป็นแบบนี้:

  } catch (err) {
    if (isIgnorableInteractionError(err)) {
      console.warn(`⚠️ Ignored interaction error: ${err.code} ${err.message}`);
      return;
    }

    logDiscordError('❌ Interaction error:', err);

    const msg = {
      content: '❌ เกิดข้อผิดพลาด กรุณาลองใหม่',
      flags: 64,
    };

    try {
      if (interaction.replied || interaction.deferred) {
        await interaction.followUp(msg);
      } else {
        await interaction.reply(msg);
      }
    } catch (replyError) {
      if (!isIgnorableInteractionError(replyError)) {
        logDiscordError('❌ Failed to report interaction error:', replyError);
      }
    }
  }

ผลที่ได้:

ไม่ใช้ ephemeral: true แล้ว
ไม่ spam log เวลา interaction หมดอายุ
ไม่ dump callback URL ยาว ๆ ลง log
กัน error ซ้ำจากการพยายามตอบ interaction ที่หมดอายุแล้ว
4. เช็กว่า bot ไม่รันซ้อนหลาย instance

เป้าหมาย: แก้สาเหตุใหญ่ของ Unknown interaction ที่ขึ้นเป็นชุด

จาก log มันขึ้นหลายคำสั่งในเวลาเดียวกัน เช่น /panel, showModal, /api-status อาการแบบนี้มีโอกาสสูงว่า bot อาจรันซ้อน หรือ interaction ถูก handler มากกว่า 1 ที่

เช็กใน Render/hosting:

เปิด service ที่รัน bot
ดูว่า instance count เป็น 1
ปิด autoscale ถ้ามี
เช็กว่าไม่มี service อื่นใช้ DISCORD_BOT_TOKEN ตัวเดียวกัน
เช็กว่าไม่ได้รัน bot ทั้งใน Web Service และ Worker พร้อมกัน
เช็ก local เครื่องตัวเองว่าไม่ได้เปิด npm run dev ทิ้งไว้พร้อมกับ production

ถ้าใช้ Render แนะนำโครงสร้างแบบนี้:

1 bot token = 1 running process เท่านั้น

ห้ามมี:

Render Web Service รัน bot
+ Render Worker รัน bot
+ local npm run dev

พร้อมกันด้วย token เดียว

5. เพิ่ม interaction dedupe ด้วย interaction.id

เป้าหมาย: กัน interaction เดิมถูก process ซ้ำใน process เดียว

เพิ่มไว้บน ๆ ของ bot/src/index.js หลังตัวแปร global เช่นใกล้ let shuttingDown = false;:

const seenInteractions = new Set();

function markInteractionSeen(id) {
  if (seenInteractions.has(id)) return false;

  seenInteractions.add(id);

  setTimeout(() => {
    seenInteractions.delete(id);
  }, 60_000).unref?.();

  return true;
}

แล้วใน event นี้:

client.on('interactionCreate', async (interaction) => {

ให้ใส่บรรทัดนี้เป็นบรรทัดแรก:

  if (!markInteractionSeen(interaction.id)) return;

รวมแล้วจะเป็น:

client.on('interactionCreate', async (interaction) => {
  if (!markInteractionSeen(interaction.id)) return;

  try {
    if (interaction.isModalSubmit()) {
      if (interaction.customId.startsWith('run_modal:')) return run.handleModal(interaction);
      if (['panel_add_modal', 'panel_done_modal', 'panel_edit_modal', 'panel_delete_modal'].includes(interaction.customId)) {
        return panel.handlePanelModal(interaction);
      }
      return;
    }

    // โค้ดเดิมต่อ...
  } catch (err) {
    // catch ใหม่จากข้อ 2
  }
});
